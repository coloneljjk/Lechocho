<script>
let leadCounter = 1;
let currentConnectedPhone = '';
let currentLeadId = null;
let currentRealPhone = '';

// Soumission d'une commande client
document.getElementById('clientOrderForm').onsubmit = (e) => {
  e.preventDefault();
  leadCounter++;
  const name = document.getElementById('newClientName').value;
  const phone = document.getElementById('newClientPhone').value;
  const car = document.getElementById('newClientCar').value;

  const container = document.getElementById('ordersContainer');
  const newCard = document.createElement('article');
  newCard.className = 'order-card';
  newCard.id = 'lead-' + leadCounter;
  newCard.setAttribute('data-owner-phone', phone);

  const showDelete = (currentConnectedPhone === phone) ? '' : 'hidden';

  newCard.innerHTML = `
    <h4>${car}</h4>
    <p style="color:#64748b; font-size:13px;">Acheteur : ${name}<br>Téléphone : <span class="phone-display" style="background:#f1f5f9; padding:2px 6px; font-weight:bold;">••• ••• •• •• 🔒</span></p>
    <button class="btn dark" onclick="openPaymentModal(${leadCounter}, '${car}', '${phone}')">🔓 Débloquer (500 FCFA)</button>
    <button class="btn danger owner-delete-btn ${showDelete}" style="margin-top:8px;" onclick="deleteOrder(${leadCounter})">🗑️ Supprimer ma commande</button>
  `;

  container.prepend(newCard);
  alert("Commande publiée avec succès !");
  e.target.reset();
};

// Suppression de commande
function deleteOrder(id) {
  if(confirm("Voulez-vous supprimer cette commande ?")) {
    document.getElementById('lead-' + id).remove();
  }
}

// Modals
function openPaymentModal(id, car, phone) {
  currentLeadId = id;
  currentRealPhone = phone;
  document.getElementById('modalVehicleName').textContent = "Véhicule : " + car;
  document.getElementById('paymentModal').classList.remove('hidden');
}

function closePaymentModal() {
  document.getElementById('paymentModal').classList.add('hidden');
}

function openAdminModal() {
  document.getElementById('adminModal').classList.remove('hidden');
}

function closeAdminModal() {
  document.getElementById('adminModal').classList.add('hidden');
}

// Paiement simulé
function processPayment() {
  closePaymentModal();
  const card = document.getElementById('lead-' + currentLeadId);
  if(card) {
    const span = card.querySelector('.phone-display');
    span.style.background = '#dcfce7';
    span.style.color = '#166534';
    span.textContent = '📞 ' + currentRealPhone;
  }
  alert("Paiement réussi de 500 FCFA ! Numéro débloqué.");
}

// Connexion vendeur simulée
document.getElementById('loginForm').onsubmit = (e) => {
  e.preventDefault();
  const phone = document.getElementById('loginPhone').value;
  currentConnectedPhone = phone;
  document.getElementById('loginForm').classList.add('hidden');
  document.getElementById('loggedInArea').classList.remove('hidden');
  document.getElementById('loggedPhone').textContent = phone;

  // Affiche les boutons de suppression si le vendeur est le propriétaire de la commande
  document.querySelectorAll('.order-card').forEach(card => {
    if(card.getAttribute('data-owner-phone') === phone) {
      card.querySelector('.owner-delete-btn').classList.remove('hidden');
    }
  });

  alert("Connecté avec succès !");
};

document.getElementById('btnLogout').onclick = () => {
  currentConnectedPhone = '';
  document.getElementById('loggedInArea').classList.add('hidden');
  document.getElementById('loginForm').classList.remove('hidden');
  document.querySelectorAll('.owner-delete-btn').forEach(btn => btn.classList.add('hidden'));
  document.getElementById('loginForm').reset();
};
</script>
